
6. DHT11
================

需要手动安装 ``DHT sensor library `` 和 ``Arduino_RouterBridge ``库

DHT11接2引脚

.. code-block:: arduino

    // DHT11 on pin 2 (Arduino App Lab - optimized version)

    #include <Arduino_RouterBridge.h>
    #include "DHT.h"

    #define DHTPIN 2
    #define DHTTYPE DHT11

    DHT dht(DHTPIN, DHTTYPE);

    // Timing control (non-blocking)
    unsigned long lastReadTime = 0;
    const unsigned long interval = 2000; // DHT11 needs ≥2s

    void setup() {
        Monitor.begin();
        dht.begin();
    }

    void loop() {
        unsigned long currentMillis = millis();

        // Read sensor every 2 seconds (non-blocking)
        if (currentMillis - lastReadTime >= interval) {
            lastReadTime = currentMillis;

            float humidity = dht.readHumidity();
            float temperature = dht.readTemperature(); // Celsius

            if (isnan(humidity) || isnan(temperature)) {
                Monitor.println("Failed to read from DHT11!");
                return;
            }

            Monitor.print("Temp: ");
            Monitor.print(temperature);
            Monitor.print(" °C  ");

            Monitor.print("Humidity: ");
            Monitor.print(humidity);
            Monitor.println(" %");
        }
    }
